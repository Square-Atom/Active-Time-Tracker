# Build-time dependencies (to produce the standalone app).
# Install with:  python -m pip install -r requirements-build.txt
-r requirements.txt

# Windows is built with Nuitka (see buildwin.py for why not PyInstaller).
# Pinned for the same reason PyInstaller is: a packaging tool that changes
# under a release can break the app, or get it flagged by antivirus again.
# Before raising it, build, run the exe from a mapped network drive, and look
# its hash up on VirusTotal.
nuitka==4.2.1 ; sys_platform == "win32"

# macOS and Linux are still built with PyInstaller. Pinned since v1.5.0 picked
# up 6.22.1 the day it appeared, whose onefile bootloader refused to start the
# Windows exe from a mapped network drive ("Security validation failure: parent
# process has different executable!"). Bump deliberately, testing the build.
pyinstaller==6.22.0 ; sys_platform != "win32"
